# Optional: Zero-Shot Auto-Annotation (Grounding DINO via Autodistill)
# --------------------------------------------------------------------------
# NOT required for the main application. Install only if you want to use
# Dataset > Auto-Annotate (Zero-Shot)... to pre-label images automatically
# from text prompts, before training your own YOLO model.
#
#   pip install -r requirements-autoannotate.txt
#
# Heavy: pulls in PyTorch + the GroundingDINO weights (downloaded on first
# use). A CUDA GPU is strongly recommended - CPU inference works but is
# slow (several seconds per image).
# --------------------------------------------------------------------------
autodistill>=0.1.0
autodistill-grounding-dino>=0.1.0
supervision>=0.20.0

# autodistill imports roboflow internally but does not always pull it in as
# a dependency, which surfaces as "ModuleNotFoundError: No module named
# 'roboflow'" the first time the annotator is loaded. Install it explicitly.
roboflow>=1.1.0

# GroundingDINO's bertwarper.py calls legacy BERT helpers
# (get_extended_attention_mask, get_head_mask) that transformers 5.x
# removed or re-signed, so loading the model fails with
# "'BertModel' object has no attribute 'get_extended_attention_mask'".
# Keep transformers on the 4.x line until GroundingDINO supports 5.x.
transformers>=4.36,<5

# NOTE: if the annotator fails with
#   ImportError: cannot import name 'BertModel' from 'transformers'
# check whether torchao got pulled in. transformers only guards its
# torchao quantizer with is_torchao_available(), which tests that the
# package exists - not that it imports - so a torchao built for a newer
# torch (e.g. needing torch.utils._pytree.register_constant, added after
# torch 2.6) takes the whole transformers import down with it. Nothing
# here uses quantization, so simply remove it:
#   pip uninstall -y torchao
